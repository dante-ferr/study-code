# input: list1, n (tamanho da list1), list2, m (tamanho da list2)
# tanto list1 quanto list2 estão ordenadas

list3 array[n + m] int

i = 1
j = 1

while i <= n and j <= m do
    if list1[i] < list2[j]
        list3[i + j] = list1[i]
        i += 1
    else
        list3[i + j] = list2[j]
        j += 1
