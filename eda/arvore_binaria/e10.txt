function search(node, parent, value)
    if value < node.value then
        res_node, res_parent = search(node.left, node, value)
    else if value > node.value then
        res_node, res_parent = search(node.right, node, value)
    else 
        res_node = node
        res_parent = parent

    return res_node, res_parent
